\begin{document}
\begin{tikzpicture}[node distance=6mm]
  \node[tahap]                          (data)  {Pengumpulan\\data};
  \node[tahap, right=of data]           (pra)   {Prapemrosesan};
  \node[tahapaksen, right=of pra]       (model) {Pelatihan\\model};
  \node[tahap, right=of model]          (uji)   {Pengujian};
  \node[tahap, right=of uji]            (nilai) {Evaluasi\\hasil};

  \draw[panah]      (data)  -- (pra);
  \draw[panah]      (pra)   -- (model);
  \draw[panahaksen] (model) -- (uji);
  \draw[panah]      (uji)   -- (nilai);

  % Kotak putus-putus mengelompokkan tahap yang diulang sampai hasil memadai.
  \begin{scope}[on background layer]
    \node[kelompok, fit=(model)(uji), label={[font=\scriptsize, text=netral]below:iterasi}] {};
  \end{scope}
\end{tikzpicture}
